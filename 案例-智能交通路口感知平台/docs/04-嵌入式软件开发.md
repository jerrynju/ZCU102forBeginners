## 3. Linux 驱动开发

### 3.1 视频帧采集驱动（V4L2）

```c
/* software/drivers/traffic_video/traffic_video.c
 * 自定义 V4L2 驱动，管理 VDMA 帧缓冲区
 * 应用层通过 /dev/video0-3 读取各路摄像头帧 */

#include <linux/module.h>
#include <linux/platform_device.h>
#include <media/v4l2-device.h>
#include <media/v4l2-ioctl.h>
#include <media/videobuf2-dma-contig.h>

struct traffic_video_dev {
    struct v4l2_device    v4l2_dev;
    struct video_device   vdev;
    struct vb2_queue      queue;
    struct mutex          lock;
    /* VDMA 寄存器基地址（从设备树获取）*/
    void __iomem         *vdma_base;
    /* 帧缓冲区物理地址（来自 reserved-memory）*/
    dma_addr_t            frame_base;
    int                   channel;
};

/* VDMA 寄存器操作：配置目标地址和帧尺寸 */
static void vdma_set_frame(struct traffic_video_dev *dev,
                           dma_addr_t addr, u32 width, u32 height) {
    /* VDMA S2MM (Stream to Memory Map) 方向 */
    iowrite32(addr,   dev->vdma_base + VDMA_S2MM_START_ADDR);
    iowrite32(width,  dev->vdma_base + VDMA_S2MM_HSIZE);
    iowrite32(height, dev->vdma_base + VDMA_S2MM_VSIZE);
    iowrite32(VDMA_S2MM_DMACR_RUN, dev->vdma_base + VDMA_S2MM_DMACR);
}

/* V4L2 QBUF / DQBUF 操作 — 零拷贝：直接映射 DDR 帧缓冲区 */
static int queue_setup(struct vb2_queue *q, unsigned int *nbuffers,
                       unsigned int *nplanes, unsigned int sizes[],
                       struct device *alloc_devs[]) {
    struct traffic_video_dev *dev = vb2_get_drv_priv(q);
    /* NV12 格式：Y 平面 + UV 平面 */
    sizes[0] = dev->width * dev->height * 3 / 2;
    *nplanes = 1;
    if (*nbuffers < 3) *nbuffers = 3;  /* 至少 3 帧缓冲（双缓冲+1）*/
    return 0;
}
```

### 4.1 主循环设计（多线程流水线）

```cpp
// software/app/main.cpp
#include <thread>
#include <atomic>
#include "frame_manager.h"
#include "inference_engine.h"
#include "cloud_reporter.h"

int main() {
    Config config = Config::load("/etc/traffic_edge/config.toml");
    FrameManager frame_mgr(NUM_CHANNELS);
    InferenceEngine infer_engine;
    CloudReporter reporter(config.mqtt);
    TrafficTracker tracker;
    EventDetector event_det(config.rules);

    // 线程 1-4：各路摄像头帧同步（从 V4L2 读取就绪帧）
    std::vector<std::thread> capture_threads;
    for (int ch = 0; ch < NUM_CHANNELS; ch++) {
        capture_threads.emplace_back([&, ch]() {
            V4L2Capture cap("/dev/video" + std::to_string(ch));
            while (!g_quit) {
                auto frame = cap.dqbuf();      // 零拷贝，返回物理地址
                frame_mgr.push(ch, frame);
                cap.qbuf(frame);
            }
        });
    }

    // 线程 5：推理调度（轮询 4 路，送 DPU）
    std::thread infer_thread([&]() {
        while (!g_quit) {
            for (int ch = 0; ch < NUM_CHANNELS; ch++) {
                auto frame = frame_mgr.pop(ch, /*timeout_ms=*/50);
                if (!frame) continue;
                // DPU 推理（YOLOv8 + LPRNet）
                auto result = infer_engine.run(frame->data, ch);
                // 多目标跟踪
                tracker.update(result);
                // 事件检测（闯红灯/逆行等）
                auto events = event_det.detect(result, traffic_state);
                // 写 OSD（发给 PL OSD IP 的寄存器）
                osd_writer.write(ch, result);
                // 上报（放入异步队列）
                if (!events.empty())
                    reporter.enqueue(events, frame->thumbnail_jpeg());
                frame_mgr.release(frame);
            }
        }
    });

    // 线程 6：云端上报（异步 MQTT）
    std::thread report_thread([&]() { reporter.run(); });

    // 主线程：gRPC 服务（阻塞）
    GrpcServer grpc_server(config.grpc_port);
    grpc_server.run();  // 阻塞直到收到退出信号
    g_quit = true;
    for (auto& t : capture_threads) t.join();
    infer_thread.join();
    report_thread.join();
}
```

### 4.2 违规事件检测

```cpp
// software/app/event_detector.cpp

class EventDetector {
public:
    struct TrafficRule {
        std::vector<cv::Point> stop_line;    // 停车线多边形
        std::vector<cv::Point> crosswalk;   // 人行横道
        std::string signal_topic;            // 订阅的信号灯状态
    };

    std::vector<TrafficEvent> detect(
        const DetectionResult_t& result,
        const SignalState& signal) {

        std::vector<TrafficEvent> events;

        for (int i = 0; i < result.object_count; i++) {
            auto& obj = result.objects[i];

            // 闯红灯检测：红灯状态下车辆越过停车线
            if (signal.is_red && obj.class_id <= 2) {  // 机动车
                cv::Point center(obj.x + obj.w/2, obj.y + obj.h);
                if (cv::pointPolygonTest(rules_.stop_line, center, false) > 0) {
                    if (!already_crossing_[obj.track_id]) {
                        events.push_back({
                            .type      = EventType::RED_LIGHT_VIOLATION,
                            .track_id  = obj.track_id,
                            .plate     = std::string(obj.plate),
                            .timestamp = result.timestamp_us,
                            .channel   = result.channel,
                            .location  = center,
                        });
                        already_crossing_[obj.track_id] = true;
                    }
                }
            }

            // 逆行检测：轨迹方向与道路方向相反
            auto motion = tracker_.get_motion_vector(obj.track_id);
            if (motion.has_value()) {
                float angle = std::atan2(motion->y, motion->x) * 180 / M_PI;
                float expected = rules_.expected_direction_deg;
                if (angle_diff(angle, expected) > 150.f) {  // 方向偏差 > 150°
                    events.push_back({.type = EventType::WRONG_WAY_DRIVING, ...});
                }
            }
        }
        return events;
    }

private:
    std::unordered_map<int, bool> already_crossing_;
};
```

### 4.3 MQTT 云端上报协议

```cpp
// software/app/cloud_reporter.cpp

class CloudReporter {
    mosquitto* mosq_;
    const std::string STATS_TOPIC   = "traffic/edge/{device_id}/stats";
    const std::string EVENT_TOPIC   = "traffic/edge/{device_id}/events";
    const std::string STATUS_TOPIC  = "traffic/edge/{device_id}/status";

public:
    void publish_stats(const FlowStats& stats) {
        // 每分钟上报流量统计
        nlohmann::json j = {
            {"ts",        stats.timestamp},
            {"device_id", config_.device_id},
            {"channels",  {}},
        };
        for (int ch = 0; ch < 4; ch++) {
            j["channels"].push_back({
                {"ch",      ch},
                {"car",     stats.counts[ch].car},
                {"truck",   stats.counts[ch].truck},
                {"bus",     stats.counts[ch].bus},
                {"person",  stats.counts[ch].person},
                {"speed_avg", stats.speed_avg[ch]},  // km/h
            });
        }
        auto payload = j.dump();
        mosquitto_publish(mosq_, nullptr,
            fmt::format(STATS_TOPIC, config_.device_id).c_str(),
            payload.size(), payload.data(), 1, false);  // QoS 1
    }

    void publish_event(const TrafficEvent& ev, std::vector<uint8_t> jpeg) {
        // 违规事件：带证据图片（Base64 或独立上传 OSS）
        nlohmann::json j = {
            {"type",      event_type_str(ev.type)},
            {"plate",     ev.plate},
            {"ts",        ev.timestamp},
            {"channel",   ev.channel},
            {"image_url", upload_to_oss(jpeg)},  // 上传图片到对象存储
        };
        auto payload = j.dump();
        mosquitto_publish(mosq_, nullptr,
            fmt::format(EVENT_TOPIC, config_.device_id).c_str(),
            payload.size(), payload.data(), 1, true);  // QoS 1, retain
    }
};
```

// software/app/rest_server.cpp - WebSocket 视频推流
ws_server.on_message([&](auto* ws, std::string_view msg, ...) {
    auto req = nlohmann::json::parse(msg);
    int channel = req["channel"];

    // 启动视频推送线程：每 100ms 推一帧（~10fps，Web 预览用）
    push_threads_[channel] = std::thread([ws, channel, &frame_mgr]() {
        while (ws->isSubscribed()) {
            auto frame = frame_mgr.get_latest(channel);
            // 压缩为 JPEG（quality=60，降低带宽）
            std::vector<uint8_t> jpeg;
            cv::imencode(".jpg", frame->bgr_preview, jpeg,
                         {cv::IMWRITE_JPEG_QUALITY, 60});
            // Base64 编码后通过 WebSocket 发送
            ws->send(base64_encode(jpeg));
            std::this_thread::sleep_for(100ms);
        }
    });
});
