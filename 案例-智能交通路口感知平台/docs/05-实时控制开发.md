# 05 - 实时控制开发（R5 FreeRTOS + OpenAMP + CAN）

## 1. 设计目标

Cortex-R5 运行 FreeRTOS，独立于 A53 Linux，承担：
- **信号灯状态机**：根据 A53 下发的控制指令管理各方向灯序
- **CAN 总线通信**：与信号灯控制器、地感线圈收发数据
- **硬件看门狗**：监控 A53 Linux 心跳，故障时切换安全状态
- **车辆计数中断**：实时响应地感线圈脉冲，精度 < 1ms

## 2. FreeRTOS 任务设计

```c
/* software/r5_freertos/main.c */
#include "FreeRTOS.h"
#include "task.h"
#include "semphr.h"
#include "queue.h"
#include "openamp/open_amp.h"

/* 任务优先级（数字越大优先级越高）*/
#define PRIORITY_WATCHDOG    5   /* 最高：看门狗监控 */
#define PRIORITY_CAN_RX      4   /* CAN 接收（时间敏感）*/
#define PRIORITY_SIGNAL_FSM  3   /* 信号灯状态机 */
#define PRIORITY_OPENAMP     2   /* 与 A53 通信 */
#define PRIORITY_TELEMETRY   1   /* 遥测数据上报 */

/* 任务句柄 */
static TaskHandle_t h_watchdog, h_can_rx, h_signal_fsm;
static TaskHandle_t h_openamp, h_telemetry;

/* 跨任务通信 */
static QueueHandle_t q_can_rx_msg;    /* CAN 消息队列 */
static QueueHandle_t q_ctrl_cmd;      /* A53 → R5 控制指令 */
static SemaphoreHandle_t sem_pps;     /* GPS PPS 时间同步信号量 */

int main(void) {
    /* 初始化硬件 */
    hw_init();
    can_init();
    openamp_init();

    /* 创建任务 */
    xTaskCreate(task_watchdog,   "WDT", 256, NULL, PRIORITY_WATCHDOG,  &h_watchdog);
    xTaskCreate(task_can_rx,     "CAN", 512, NULL, PRIORITY_CAN_RX,    &h_can_rx);
    xTaskCreate(task_signal_fsm, "FSM", 1024,NULL, PRIORITY_SIGNAL_FSM,&h_signal_fsm);
    xTaskCreate(task_openamp,    "AMP", 512, NULL, PRIORITY_OPENAMP,   &h_openamp);
    xTaskCreate(task_telemetry,  "TEL", 256, NULL, PRIORITY_TELEMETRY, &h_telemetry);

    vTaskStartScheduler();
    /* 不应到达这里 */
    for(;;);
}
```

## 3. 信号灯状态机

```c
/* software/r5_freertos/signal_fsm.c */

/* 路口信号灯状态（标准四相位）*/
typedef enum {
    PHASE_NS_GREEN  = 0,  /* 南北直行绿灯 */
    PHASE_NS_YELLOW = 1,  /* 南北黄灯 */
    PHASE_EW_GREEN  = 2,  /* 东西直行绿灯 */
    PHASE_EW_YELLOW = 3,  /* 东西黄灯 */
    PHASE_NS_LEFT   = 4,  /* 南北左转绿灯 */
    PHASE_NS_LEFT_Y = 5,  /* 南北左转黄灯 */
    PHASE_EW_LEFT   = 6,  /* 东西左转绿灯 */
    PHASE_EW_LEFT_Y = 7,  /* 东西左转黄灯 */
    PHASE_ALLRED    = 8,  /* 全红（相位间隔）*/
} SignalPhase_t;

typedef struct {
    SignalPhase_t current_phase;
    uint32_t      phase_timer_ms;  /* 当前相位已运行时间 */
    uint32_t      phase_duration[9]; /* 各相位时长（可动态调整）*/
    bool          adaptive_enable;   /* 是否启用自适应控制 */
    uint32_t      queue_length[4];   /* A53 上报的各方向排队长度 */
} SignalFSM_t;

static SignalFSM_t g_fsm = {
    .current_phase = PHASE_NS_GREEN,
    .phase_duration = {
        [PHASE_NS_GREEN]  = 45000,  /* ms */
        [PHASE_NS_YELLOW] = 3000,
        [PHASE_EW_GREEN]  = 45000,
        [PHASE_EW_YELLOW] = 3000,
        [PHASE_NS_LEFT]   = 20000,
        [PHASE_NS_LEFT_Y] = 3000,
        [PHASE_EW_LEFT]   = 20000,
        [PHASE_EW_LEFT_Y] = 3000,
        [PHASE_ALLRED]    = 2000,
    },
};

void task_signal_fsm(void *arg) {
    TickType_t xLastWakeTime = xTaskGetTickCount();
    const TickType_t xPeriod = pdMS_TO_TICKS(10);  /* 10ms 执行周期 */

    for (;;) {
        /* 接收 A53 的控制指令（非阻塞）*/
        CtrlCmd_t cmd;
        if (xQueueReceive(q_ctrl_cmd, &cmd, 0) == pdTRUE) {
            process_ctrl_cmd(&g_fsm, &cmd);
        }

        /* 更新状态机 */
        g_fsm.phase_timer_ms += 10;

        /* 自适应控制：根据排队长度动态调整绿灯时长 */
        if (g_fsm.adaptive_enable) {
            adaptive_timing_adjust(&g_fsm);
        }

        /* 相位切换判断 */
        if (g_fsm.phase_timer_ms >= g_fsm.phase_duration[g_fsm.current_phase]) {
            g_fsm.phase_timer_ms = 0;
            g_fsm.current_phase  = next_phase(g_fsm.current_phase);
            /* 发送 CAN 帧，控制实际信号灯硬件 */
            can_send_signal_cmd(g_fsm.current_phase);
        }

        /* 精确周期等待（避免累积误差）*/
        vTaskDelayUntil(&xLastWakeTime, xPeriod);
    }
}

/* 自适应时长调整（简化版韦伯斯特算法）*/
static void adaptive_timing_adjust(SignalFSM_t* fsm) {
    /* 仅在当前相位开始时计算（避免频繁变化）*/
    if (fsm->phase_timer_ms > 100) return;

    /* 南北方向排队比东西方向多 20% 则延长南北绿灯 */
    uint32_t ns_queue = fsm->queue_length[0] + fsm->queue_length[2];
    uint32_t ew_queue = fsm->queue_length[1] + fsm->queue_length[3];
    uint32_t total = ns_queue + ew_queue;
    if (total == 0) return;

    uint32_t cycle = 90000; /* 总周期 90s */
    uint32_t min_green = 15000, max_green = 60000;
    uint32_t ns_green = cycle * ns_queue / total;
    ns_green = CLAMP(ns_green, min_green, max_green);

    fsm->phase_duration[PHASE_NS_GREEN] = ns_green;
    fsm->phase_duration[PHASE_EW_GREEN] = cycle - ns_green - 4*3000 - 2*2000;
}
```

## 4. CAN 总线通信

```c
/* software/r5_freertos/can_driver.c */

/* CAN 帧 ID 定义（11bit 标准帧）*/
#define CAN_ID_SIGNAL_CMD     0x101  /* → 信号灯控制器 */
#define CAN_ID_SIGNAL_STATUS  0x102  /* ← 信号灯状态反馈 */
#define CAN_ID_LOOP_DETECTOR  0x200  /* ← 地感线圈计数脉冲 */
#define CAN_ID_HEARTBEAT      0x7FF  /* → 心跳帧 */

/* CAN 波特率：500 Kbps（交通控制标准）*/
#define CAN_BITRATE_500K  0x00230116  /* 50MHz 时钟，500K */

void can_init(void) {
    /* 配置 AXI CAN 控制器 */
    XCanPs_Config *cfg = XCanPs_LookupConfig(XPAR_PSU_CAN_0_DEVICE_ID);
    XCanPs_CfgInitialize(&g_can, cfg, cfg->BaseAddr);
    XCanPs_SetBitTiming(&g_can, 3, 1, 22); /* 500Kbps @ 50MHz */
    XCanPs_SetHandler(&g_can, XCANPS_HANDLER_RECV, can_rx_handler, NULL);
    XCanPs_EnterMode(&g_can, XCANPS_MODE_NORMAL);
}

/* CAN 接收中断处理（在中断上下文中，必须快速返回）*/
void can_rx_handler(void *arg) {
    u32 frame[4];
    if (XCanPs_Recv(&g_can, frame) == XST_SUCCESS) {
        u32 id = XCANPS_GET_ID_IDT1(frame[0]) >> 21;
        CANMsg_t msg = {.id = id, .dlc = XCANPS_GET_DLC(frame[1])};
        memcpy(msg.data, &frame[2], msg.dlc);
        /* 发送到队列，由 task_can_rx 处理（不在中断中做复杂操作）*/
        BaseType_t woken = pdFALSE;
        xQueueSendFromISR(q_can_rx_msg, &msg, &woken);
        portYIELD_FROM_ISR(woken);
    }
}

void task_can_rx(void *arg) {
    CANMsg_t msg;
    for (;;) {
        if (xQueueReceive(q_can_rx_msg, &msg, portMAX_DELAY) == pdTRUE) {
            switch (msg.id) {
            case CAN_ID_SIGNAL_STATUS:
                /* 信号灯控制器状态反馈（确认指令执行）*/
                process_signal_feedback(msg.data);
                break;
            case CAN_ID_LOOP_DETECTOR:
                /* 地感线圈：bit0-3 对应 4 个方向，高电平表示有车 */
                process_loop_detector(msg.data[0]);
                break;
            }
        }
    }
}

/* 发送信号灯控制帧 */
void can_send_signal_cmd(SignalPhase_t phase) {
    u32 frame[4] = {0};
    XCANPS_SET_ID1(frame[0], CAN_ID_SIGNAL_CMD << 21);  /* 设置帧 ID */
    XCANPS_SET_DLC(frame[1], 2);                          /* 数据长度 2 字节 */
    frame[2] = (phase << 8) | compute_checksum(phase);   /* 相位 + 校验 */
    XCanPs_Send(&g_can, frame);
}
```

## 5. OpenAMP 与 A53 Linux 通信

```c
/* software/r5_freertos/openamp_rpmsg.c
 * R5 通过 RPMsg 接收 A53 的控制指令，上报状态和车辆计数 */

#include "openamp/open_amp.h"
#include "metal/alloc.h"

#define RPMSG_ENDPOINT_NAME  "traffic-ctrl"
#define RPMSG_ADDR           0x400

/* 消息类型定义（与 A53 侧共享头文件）*/
typedef enum {
    MSG_TYPE_CTRL_CMD    = 0x01,  /* A53 → R5：控制指令 */
    MSG_TYPE_PHASE_QUERY = 0x02,  /* A53 → R5：查询当前相位 */
    MSG_TYPE_STATUS_RESP = 0x03,  /* R5 → A53：状态响应 */
    MSG_TYPE_COUNT_REPORT= 0x04,  /* R5 → A53：车辆计数上报 */
    MSG_TYPE_HEARTBEAT   = 0xFF,  /* 双向心跳 */
} RPMsgType_t;

typedef struct __attribute__((packed)) {
    uint8_t  type;
    uint8_t  seq;
    uint16_t length;
    uint8_t  payload[60];
} RPMsgFrame_t;

/* RPMsg 回调：A53 发来消息时触发 */
static int rpmsg_rx_callback(struct rpmsg_endpoint *ept,
                              void *data, size_t len,
                              uint32_t src, void *priv) {
    RPMsgFrame_t *frame = (RPMsgFrame_t *)data;
    switch (frame->type) {
    case MSG_TYPE_CTRL_CMD: {
        CtrlCmd_t cmd;
        memcpy(&cmd, frame->payload, sizeof(cmd));
        xQueueSend(q_ctrl_cmd, &cmd, 0);
        break;
    }
    case MSG_TYPE_HEARTBEAT:
        /* 重置看门狗计时器 */
        g_a53_heartbeat_timeout = 0;
        /* 回复心跳 */
        rpmsg_send_heartbeat();
        break;
    }
    return 0;
}

/* 每秒上报一次车辆计数到 A53 */
void rpmsg_report_counts(void) {
    RPMsgFrame_t frame = {.type = MSG_TYPE_COUNT_REPORT, .seq = g_seq++};
    CountReport_t *report = (CountReport_t *)frame.payload;
    report->timestamp = get_r5_timestamp();
    memcpy(report->counts, g_vehicle_counts, sizeof(g_vehicle_counts));
    frame.length = sizeof(CountReport_t);
    rpmsg_send(g_ept, &frame, sizeof(RPMsgFrame_t));
    /* 清零计数（上报后重置）*/
    memset(g_vehicle_counts, 0, sizeof(g_vehicle_counts));
}
```

## 6. 硬件看门狗与故障安全

```c
/* software/r5_freertos/watchdog.c */

/* 看门狗策略：
 * - 软件看门狗（FreeRTOS 任务）：监控各任务心跳
 * - 硬件看门狗（LPD_SWDT）：监控整个 R5 子系统
 * - A53 心跳超时：A53 Linux 崩溃时，R5 切换到安全模式 */

#define A53_HEARTBEAT_TIMEOUT_MS 5000  /* 5s 无心跳视为 A53 故障 */
#define SAFE_MODE_PHASE  PHASE_ALLRED  /* 安全模式：全红 */

static volatile uint32_t g_a53_heartbeat_timeout = 0;

void task_watchdog(void *arg) {
    /* 初始化硬件看门狗（60s 超时）*/
    XWdtPs wdt;
    XWdtPs_Config *cfg = XWdtPs_LookupConfig(XPAR_PSU_WDT_0_DEVICE_ID);
    XWdtPs_CfgInitialize(&wdt, cfg, cfg->BaseAddr);
    XWdtPs_SetControlValues(&wdt, XWDTPS_COUNTER_RSTVAL, 0xFFF);
    XWdtPs_Start(&wdt);

    for (;;) {
        /* 喂硬件看门狗 */
        XWdtPs_RestartWdt(&wdt);

        /* 监控 A53 心跳 */
        g_a53_heartbeat_timeout += 100;
        if (g_a53_heartbeat_timeout > A53_HEARTBEAT_TIMEOUT_MS) {
            /* A53 疑似宕机，切换安全模式（全红灯）*/
            g_fsm.current_phase = SAFE_MODE_PHASE;
            can_send_signal_cmd(SAFE_MODE_PHASE);
            g_fsm.adaptive_enable = false;  /* 禁用自适应 */
            /* 尝试通过 RPMsg 告知 A53（可能已无响应，忽略失败）*/
            rpmsg_send_alert(ALERT_A53_HEARTBEAT_TIMEOUT);
        }

        /* 监控 CAN 总线健康（信号灯控制器无响应）*/
        if (g_can_response_timeout > 3000) {
            rpmsg_send_alert(ALERT_CAN_CONTROLLER_OFFLINE);
        }

        vTaskDelay(pdMS_TO_TICKS(100));
    }
}
```

## 7. GPS PPS 时间同步

```c
/* software/r5_freertos/pps_sync.c
 * GPS 每秒输出一个 PPS 脉冲，接入 PL GPIO，触发中断
 * R5 用于精确对齐各路摄像头帧的时间戳 */

void pps_irq_handler(void *arg) {
    /* 记录 PPS 时刻的 R5 本地时间 */
    g_pps_local_time = get_r5_cycle_count();
    g_pps_utc_second++;
    /* 通知等待时间同步的任务 */
    BaseType_t woken = pdFALSE;
    xSemaphoreGiveFromISR(sem_pps, &woken);
    portYIELD_FROM_ISR(woken);
}

uint64_t get_precise_timestamp_us(void) {
    uint64_t cycles_since_pps = get_r5_cycle_count() - g_pps_local_time;
    uint64_t us_since_pps = cycles_since_pps / (R5_FREQ_HZ / 1000000);
    return (uint64_t)g_pps_utc_second * 1000000 + us_since_pps;
}
```

## 8. A53 侧 OpenAMP 客户端

```c
/* software/app/openamp_client.c （在 Linux 用户空间运行）*/
#include <sys/ioctl.h>
#include <fcntl.h>

/* Linux 通过 /dev/rpmsg_ctrl 与 R5 通信 */
class OpenAMPClient {
    int rpmsg_fd_;

public:
    OpenAMPClient() {
        rpmsg_fd_ = open("/dev/rpmsg0", O_RDWR);
        if (rpmsg_fd_ < 0)
            throw std::runtime_error("Failed to open /dev/rpmsg0");
    }

    void send_phase_override(SignalPhase phase, int duration_ms) {
        RPMsgFrame_t frame{};
        frame.type = MSG_TYPE_CTRL_CMD;
        CtrlCmd_t cmd{.type = CMD_PHASE_OVERRIDE,
                      .phase = phase,
                      .duration_ms = duration_ms};
        memcpy(frame.payload, &cmd, sizeof(cmd));
        write(rpmsg_fd_, &frame, sizeof(frame));
    }

    void run_heartbeat_loop() {
        /* 每 1s 发送心跳，R5 用此检测 A53 健康状态 */
        while (!quit_) {
            RPMsgFrame_t hb{.type = MSG_TYPE_HEARTBEAT};
            write(rpmsg_fd_, &hb, sizeof(hb));
            std::this_thread::sleep_for(1s);
        }
    }

    CountReport_t read_counts() {
        RPMsgFrame_t frame{};
        read(rpmsg_fd_, &frame, sizeof(frame));
        CountReport_t report;
        memcpy(&report, frame.payload, sizeof(report));
        return report;
    }
};
```
