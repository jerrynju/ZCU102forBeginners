## 3. gRPC 管理接口

```protobuf
// software/proto/traffic_edge.proto
syntax = "proto3";
import "google/protobuf/timestamp.proto";

package trafficedge.v1;

// 管理控制台接口
service TrafficEdgeManagement {
    // 设备状态查询
    rpc GetDeviceStatus(GetStatusRequest) returns (DeviceStatus);

    // 实时检测事件流（Server-Streaming）
    rpc StreamDetections(StreamRequest) returns (stream DetectionEvent);

    // 历史数据查询
    rpc QueryHistory(HistoryRequest) returns (HistoryResponse);

    // 配置更新（Cloud → Edge）
    rpc UpdateConfig(UpdateConfigRequest) returns (UpdateConfigResponse);

    // 信号灯手动控制
    rpc ControlSignal(SignalControlRequest) returns (SignalControlResponse);

    // 截图触发
    rpc TriggerCapture(CaptureRequest) returns (CaptureResponse);
}

message DeviceStatus {
    string  device_id    = 1;
    google.protobuf.Timestamp ts = 2;
    string  version      = 3;
    string  hw_version   = 4;
    bool    online       = 5;
    SystemMetrics metrics = 6;
    repeated ChannelStatus channels = 7;
}

message SystemMetrics {
    float cpu_usage_pct    = 1;  // A53 平均 CPU 使用率
    float mem_usage_pct    = 2;
    float temp_celsius     = 3;  // MPSoC 结温
    float dpu_usage_pct    = 4;  // DPU 利用率
    uint64 uptime_seconds  = 5;
    string primary_link    = 6;  // "10gbe" or "4g"
    uint32 link_speed_mbps = 7;
}

message DetectionEvent {
    google.protobuf.Timestamp ts = 1;
    uint32 channel    = 2;
    uint32 frame_id   = 3;
    repeated DetectedObject objects = 4;
    bytes  preview_jpeg = 5;  // 可选：带框预览图（低频）
}

message DetectedObject {
    uint32 track_id   = 1;
    uint32 class_id   = 2;
    float  confidence = 3;
    BoundingBox bbox  = 4;
    string plate      = 5;
    float  speed_kmh  = 6;
}
```

## 4. TLS 安全通信配置

```bash
# 双向 TLS（mTLS）配置——设备与云端互相验证身份
# 每台设备烧录唯一客户端证书（在生产阶段由 CA 签发）

# 证书层次：
# Root CA（离线，硬件安全模块保存）
#   └── Intermediate CA（在线，自动签发设备证书）
#       └── Device Cert（绑定 device_id，有效期 2 年）

# MQTT 连接配置（mosquitto 客户端）
mosquitto_pub \
    --cafile /etc/ssl/certs/traffic_ca.crt \
    --cert   /etc/ssl/device/edge001.crt \
    --key    /etc/ssl/device/edge001.key \
    -h mqtt.traffic.example.com \
    -p 8883 \
    --tls-version tlsv1.3 \
    -t "traffic/edge/bj/cy/edge001/test" \
    -m "hello"
```

```cpp
// software/app/cloud_reporter.cpp — TLS 初始化
void CloudReporter::init_tls() {
    mosquitto_tls_set(mosq_,
        "/etc/ssl/certs/traffic_ca.crt",   // CA 证书
        nullptr,                            // CA 目录（不用）
        "/etc/ssl/device/edge001.crt",     // 设备证书
        "/etc/ssl/device/edge001.key",     // 设备私钥
        nullptr);                          // 密码回调（无密码保护）
    mosquitto_tls_opts_set(mosq_, 1, "tlsv1.3", nullptr);
    // 验证云端证书的 CN 字段（防中间人攻击）
    mosquitto_tls_insecure_set(mosq_, false);
}
```

## 5. 离线缓存与断线重连

```cpp
// software/app/offline_cache.cpp
// 网络断线时，将上报数据持久化到本地 SQLite，重连后自动补传

#include <sqlite3.h>

class OfflineCache {
    sqlite3 *db_;

public:
    OfflineCache(const std::string& path) {
        sqlite3_open(path.c_str(), &db_);
        sqlite3_exec(db_,
            "CREATE TABLE IF NOT EXISTS pending_msgs ("
            "  id INTEGER PRIMARY KEY AUTOINCREMENT,"
            "  topic TEXT NOT NULL,"
            "  payload BLOB NOT NULL,"
            "  qos INTEGER DEFAULT 1,"
            "  created_at INTEGER DEFAULT (strftime('%s','now'))"
            ")", nullptr, nullptr, nullptr);
    }

    void store(const std::string& topic, const std::string& payload, int qos) {
        sqlite3_stmt *stmt;
        sqlite3_prepare_v2(db_,
            "INSERT INTO pending_msgs (topic, payload, qos) VALUES (?,?,?)",
            -1, &stmt, nullptr);
        sqlite3_bind_text(stmt, 1, topic.c_str(), -1, SQLITE_STATIC);
        sqlite3_bind_blob(stmt, 2, payload.data(), payload.size(), SQLITE_STATIC);
        sqlite3_bind_int(stmt, 3, qos);
        sqlite3_step(stmt);
        sqlite3_finalize(stmt);
    }

    // 重连后批量补传（限速：不超过 1MB/s，避免冲击云端）
    void flush_to_mqtt(mosquitto *mosq) {
        const int BATCH_SIZE = 50;
        sqlite3_stmt *stmt;
        sqlite3_prepare_v2(db_,
            "SELECT id, topic, payload, qos FROM pending_msgs "
            "ORDER BY id LIMIT ?", -1, &stmt, nullptr);
        sqlite3_bind_int(stmt, 1, BATCH_SIZE);

        std::vector<int64_t> sent_ids;
        while (sqlite3_step(stmt) == SQLITE_ROW) {
            int64_t id    = sqlite3_column_int64(stmt, 0);
            auto topic    = (const char*)sqlite3_column_text(stmt, 1);
            auto payload  = sqlite3_column_blob(stmt, 2);
            int payload_len = sqlite3_column_bytes(stmt, 2);
            int qos       = sqlite3_column_int(stmt, 3);

            if (mosquitto_publish(mosq, nullptr, topic,
                    payload_len, payload, qos, false) == MOSQ_ERR_SUCCESS) {
                sent_ids.push_back(id);
            }
            std::this_thread::sleep_for(std::chrono::milliseconds(10));
        }
        sqlite3_finalize(stmt);

        // 删除已成功发送的记录
        if (!sent_ids.empty()) {
            std::string del_sql = "DELETE FROM pending_msgs WHERE id IN (";
            for (auto id : sent_ids) del_sql += std::to_string(id) + ",";
            del_sql.back() = ')';
            sqlite3_exec(db_, del_sql.c_str(), nullptr, nullptr, nullptr);
        }
    }
};
```

## 6. 4G 备份链路自动切换

```bash
# /etc/network/if-up.d/failover.sh
# 使用 ip rule 实现双链路路由：10GbE 主链路，4G 备用

# 主链路（10GbE）
ip route add default via 192.168.1.1 dev eth0 metric 100 table main

# 备用链路（4G USB modem，通过 ppp0 或 usb0）
ip route add default via 10.0.0.1 dev ppp0 metric 200 table failover

# 健康检查脚本（每 30s 检测主链路可达性）
# /usr/local/bin/link_monitor.sh
while true; do
    if ping -c 3 -W 5 -I eth0 8.8.8.8 > /dev/null 2>&1; then
        ip route change default via 192.168.1.1 dev eth0 metric 100
    else
        logger "Primary link down, switching to 4G"
        ip route change default via 10.0.0.1 dev ppp0 metric 50
        # 通知应用层（MQTT）
        mosquitto_pub -h localhost -t "system/link_failover" -m '{"link":"4g"}'
    fi
    sleep 30
done
```
